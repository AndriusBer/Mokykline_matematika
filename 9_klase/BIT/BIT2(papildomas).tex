% Spausdinti horizontaliame A4 lape, mastelis 100 %.
% Perkirpus per vidurį gaunami du statūs A5 dydžio lapeliai.
\newcommand{\remelis}[2]{%
  \tikz[baseline={([yshift=-0.6ex]current bounding box.center)}]\draw[line width=0.5pt]
    (0,0) rectangle (#1,#2);}

\newcommand{\uzduotis}[2]{%
  \begin{minipage}[t][37mm][t]{\linewidth}
    \textbf{#1.}\enspace #2\par
    \vfill
    \remelis{90mm}{9mm}\par
    \vspace{3mm}
  \end{minipage}\par
}

\newcommand{\uzduotys}[4]{%
  \uzduotis{1}{Išskaidykite dauginamaisiais: $P(x)=#1$.}
  \uzduotis{2}{Išskirkite pilnąjį kvadratą: $P(x)=#2$.}
  \uzduotis{3}{Atskliauskite naudodamiesi greitosios daugybos formule:\\$#3$.}
  \uzduotis{4}{Išskaidykite dauginamaisiais: $#4$.}
}

\newcommand{\lapelis}[2]{%
  \begin{minipage}[t][190mm][t]{128.5mm}
    \small
    \textbf{Mokinio kodas}\enspace\remelis{32mm}{8mm}%
    \hfill\textbf{#1 variantas}\par
    \vspace{6mm}
    {\centering\normalsize\textbf{Bazinių įgūdžių testas 2}\par}
    \vspace{2mm}
    {\centering\small Po užduotimis palikta vieta sprendimams.\\
      Į rėmelius įrašykite atsakymus.\par}
    \vspace{6mm}
    \normalsize
    #2
  \end{minipage}%
}
