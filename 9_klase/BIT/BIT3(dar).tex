\newcommand{\remelis}[2]{%
  \tikz[baseline={([yshift=-0.6ex]current bounding box.center)}]\draw[line width=0.5pt]
    (0,0) rectangle (#1,#2);}
\newcommand{\uzduotis}[2]{%
  \begin{minipage}[t][37mm][t]{\linewidth}
    \textbf{#1.}\enspace #2\par
    \vfill
    \remelis{90mm}{9mm}\par
    \vspace{3mm}
  \end{minipage}\par
}
\newcommand{\uzduotys}[4]{%
  \uzduotis{1}{Išskirkite pilnąjį kvadratą: $P(x)=#1$.}
  \uzduotis{2}{Išskirkite pilnąjį kvadratą: $P(x)=#2$.}
  \uzduotis{3}{Išskirkite pilnąjį kvadratą: $P(x)=#3$.}
    \uzduotis{4}{Atskliauskite naudodamiesi greitosios daugybos formule:\\$#4$.}
}
\newcommand{\lapelis}[2]{%
  \begin{minipage}[t][190mm][t]{128.5mm}
    \small
    \textbf{Mokinio kodas}\enspace\remelis{32mm}{8mm}%
    \hfill\textbf{#1 variantas}\par
    \vspace{6mm}
    {\centering\normalsize\textbf{Bazinių įgūdžių testas 3}\par}
    \vspace{2mm}
    {\centering\small Po užduotimis palikta vieta sprendimams.\\
      Į rėmelius įrašykite atsakymus.\par}
    \vspace{6mm}
    \normalsize
    #2
  \end{minipage}%
}
\newcommand{\lapas}[2]{%
  \null
  \begin{tikzpicture}[remember picture,overlay]
    \draw[dash pattern=on 2pt off 3pt,line width=0.4pt]
      (current page.north) -- (current page.south);
    \node[anchor=north west,inner sep=0pt]
      at ([xshift=10mm,yshift=-10mm]current page.north west)
      {\lapelis{I}{#1}};
    \node[anchor=north west,inner sep=0pt]
      at ([xshift=10mm,yshift=-10mm]current page.north)
      {\lapelis{II}{#2}};
  \end{tikzpicture}%
}
