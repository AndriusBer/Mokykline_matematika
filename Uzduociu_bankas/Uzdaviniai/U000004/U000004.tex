\begin{salyga}
Išspręskite lygtį:
\[
  2x+3=11.
\]
\end{salyga}

\begin{atsakymas}
$x=4$.
\end{atsakymas}

\begin{sprendimas}
Iš abiejų lygties pusių atimame $3$:
\[
  2x=11-3=8.
\]
Abi lygties puses padalijame iš $2$:
\[
  x=4.
\]
Patikriname: įstatę $x=4$, gauname $2\cdot4+3=11$, taigi lygybė teisinga.
Vadinasi, lygties sprendinys yra $x=4$.
\end{sprendimas}
