\begin{salyga}
Išspręskite lygtį:
    \[ x^3 + 2x^2 - 9x - 18 = 0 \]
\end{salyga}

\begin{atsakymas}
$x=-3$, $x=-2$ arba $x=3$.
\end{atsakymas}

\begin{sprendimas}
Grupuojame ir išskaidome:
\[
x^3+2x^2-9x-18=x^2(x+2)-9(x+2)
=(x+2)(x-3)(x+3).
\]
Sandauga lygi nuliui tada ir tik tada, kai $x=-2$, $x=3$ arba $x=-3$.
\end{sprendimas}
