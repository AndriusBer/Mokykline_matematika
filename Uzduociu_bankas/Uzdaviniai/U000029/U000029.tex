\begin{salyga}
Išspręskite lygtį:
    \[ (x^2 + 2x - 2)(x^2 + 2x + 1) = 4 \]
\end{salyga}

\begin{atsakymas}
$x=-3$ arba $x=1$.
\end{atsakymas}

\begin{sprendimas}
Tegu $t=x^2+2x$. Tada $(t-2)(t+1)=4$, taigi $t^2-t-6=(t-3)(t+2)=0$.

Jei $t=3$, gauname $x^2+2x-3=(x-1)(x+3)=0$, todėl $x=1$ arba $x=-3$. Jei $t=-2$, tai $x^2+2x+2=0$, tačiau jos diskriminantas $D=4-8=-4<0$, todėl realiųjų sprendinių nėra.
\end{sprendimas}
