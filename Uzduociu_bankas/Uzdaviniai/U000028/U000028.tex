\begin{salyga}
Išspręskite lygtį:
    \[ (x^2 - 3x)^2 - 2(x^2 - 3x) - 8 = 0 \]
\end{salyga}

\begin{atsakymas}
$x=-1$, $1$, $2$ arba $4$.
\end{atsakymas}

\begin{sprendimas}
Tegu $t=x^2-3x$. Gauname $t^2-2t-8=(t-4)(t+2)=0$.
Jei $t=4$, tai $x^2-3x-4=(x-4)(x+1)=0$, todėl $x=4$ arba $x=-1$.
Jei $t=-2$, tai $x^2-3x+2=(x-1)(x-2)=0$, todėl $x=1$ arba $x=2$.
\end{sprendimas}
