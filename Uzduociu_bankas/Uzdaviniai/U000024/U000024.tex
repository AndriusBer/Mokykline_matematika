\begin{salyga}
Išskaidykite dauginamaisiais grupavimo būdu:
    \begin{enumerate}[label=\alph*), itemsep=4pt]
        \item $x^3 - 3x^2 - 4x + 12$
        \item $2x^3 + 3x^2 - 8x - 12$
    \end{enumerate}
\end{salyga}

\begin{atsakymas}
a) $(x-3)(x-2)(x+2)$; b) $(2x+3)(x-2)(x+2)$.
\end{atsakymas}

\begin{sprendimas}
\begin{enumerate}[label=\alph*)]
\item $x^3-3x^2-4x+12=x^2(x-3)-4(x-3)=(x-3)(x^2-4)=(x-3)(x-2)(x+2)$.
\item $2x^3+3x^2-8x-12=x^2(2x+3)-4(2x+3)=(2x+3)(x^2-4)=(2x+3)(x-2)(x+2)$.
\end{enumerate}
\end{sprendimas}
