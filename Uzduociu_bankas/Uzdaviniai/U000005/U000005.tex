\begin{salyga}
Išspręskite lygtį:
\[
  3x-7=8.
\]
\end{salyga}

\begin{atsakymas}
$x=5$.
\end{atsakymas}

\begin{sprendimas}
Prie abiejų lygties pusių pridedame $7$:
\[
  3x=8+7=15.
\]
Abi lygties puses padalijame iš $3$:
\[
  x=5.
\]
Patikriname: įstatę $x=5$, gauname $3\cdot5-7=8$, taigi lygybė teisinga.
Vadinasi, lygties sprendinys yra $x=5$.
\end{sprendimas}
