\begin{salyga}
Išskaidykite kvadratinius trinarus dauginamaisiais:
    \begin{enumerate}[label=\alph*), itemsep=4pt]
        \item $2x^2 - 5x - 12$
        \item $-3x^2 + 10x - 3$
    \end{enumerate}
\end{salyga}

\begin{atsakymas}
a) $(2x+3)(x-4)$; b) $-(3x-1)(x-3)$.
\end{atsakymas}

\begin{sprendimas}
\begin{enumerate}[label=\alph*)]
\item $2x^2-5x-12=2x^2-8x+3x-12=2x(x-4)+3(x-4)=(2x+3)(x-4)$.
\item $-3x^2+10x-3=-3x^2+9x+x-3=-3x(x-3)+(x-3)=-(3x-1)(x-3)$.
\end{enumerate}
\end{sprendimas}
