\begin{salyga}
Išspręskite lygtį:
    \[ (2x - 3)^2 - 3x = (x + 2)(x - 2) + 1 \]
\end{salyga}

\begin{atsakymas}
$x=1$ arba $x=4$.
\end{atsakymas}

\begin{sprendimas}
Išskleidę skliaustus ir visus narius perkėlę į kairę gauname
\[
4x^2-15x+9=x^2-3,\qquad 3x^2-15x+12=0.
\]
Padaliję iš $3$ išskaidome: $(x-1)(x-4)=0$. Taigi $x=1$ arba $x=4$.
\end{sprendimas}
